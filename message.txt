do
    local AutoOwnerT = false
    local AutoOwnerTargetPart = nil
    local AutoOwnerGrabPart = nil
    local AutoOwnerHighlight = nil
    local AutoOwnerDestroyConn = nil

    local function getAutoOwnerRemote()
        local grabEvents = ReplicatedStorage:FindFirstChild("GrabEvents")
        if not grabEvents then return nil end
        return grabEvents:FindFirstChild("SetNetworkOwner")
    end

    local function findAutoOwnerTargetPart()
        local grabParts = Workspace:FindFirstChild("GrabParts")
        if not grabParts then return nil, nil end

        local grabPart = grabParts:FindFirstChild("GrabPart")
        if not grabPart then return nil, nil end

        local weld = grabPart:FindFirstChildOfClass("WeldConstraint")
        if not weld then return nil, nil end

        local target = weld.Part1
        if target and target:IsA("BasePart") then
            return target, grabPart
        end

        return nil, nil
    end

    local function createAutoOwnerHighlight(part)
        if not part or not part.Parent then return nil end

        local oldHighlight = part:FindFirstChild("AutoOwner_Highlight")
        if oldHighlight then
            oldHighlight:Destroy()
        end

        local highlight = Instance.new("Highlight")
        highlight.Name = "AutoOwner_Highlight"
        highlight.Adornee = part
        highlight.FillColor = Color3.fromRGB(145, 85, 255)
        highlight.FillTransparency = 0.75
        highlight.OutlineColor = Color3.fromRGB(0, 0, 0)
        highlight.OutlineTransparency = 0.35
        highlight.Parent = part
        return highlight
    end

    local function cleanupAutoOwner()
        if AutoOwnerDestroyConn then
            AutoOwnerDestroyConn:Disconnect()
            AutoOwnerDestroyConn = nil
        end

        if AutoOwnerHighlight and AutoOwnerHighlight.Parent then
            AutoOwnerHighlight:Destroy()
        end

        AutoOwnerHighlight = nil
        AutoOwnerTargetPart = nil
        AutoOwnerGrabPart = nil
        AutoOwnerT = false
    end

    local function toggleAutoOwner()
        if AutoOwnerT then
            cleanupAutoOwner()
            pcall(function()
                Rayfield:Notify({
                    Title = "자동 소유권",
                    Content = "자동 소유권이 꺼졌습니다.",
                    Duration = 2,
                    Image = 4483362458,
                })
            end)
            return
        end

        local setNetworkOwner = getAutoOwnerRemote()
        if not setNetworkOwner then
            pcall(function()
                Rayfield:Notify({
                    Title = "자동 소유권",
                    Content = "GrabEvents를 찾을 수 없습니다.",
                    Duration = 3,
                    Image = 4483362458,
                })
            end)
            return
        end

        local target, grabPart = findAutoOwnerTargetPart()

        if target then
            AutoOwnerT = true
            AutoOwnerTargetPart = target
            AutoOwnerGrabPart = grabPart
            AutoOwnerHighlight = createAutoOwnerHighlight(AutoOwnerTargetPart)

            AutoOwnerDestroyConn = AutoOwnerTargetPart.Destroying:Connect(function()
                cleanupAutoOwner()
            end)

            pcall(function()
                Rayfield:Notify({
                    Title = "자동 소유권",
                    Content = "자동 소유권이 켜졌습니다.",
                    Duration = 2,
                    Image = 4483362458,
                })
            end)
        else
            AutoOwnerT = false
            pcall(function()
                Rayfield:Notify({
                    Title = "자동 소유권",
                    Content = "물건이나 사람을 잡은 상태에서 V키를 눌러주세요.",
                    Duration = 3,
                    Image = 4483362458,
                })
            end)
        end
    end

    pcall(function()
        GrabTab:CreateSection("자동 소유권")

        if GrabTab.CreateParagraph then
            GrabTab:CreateParagraph({
                Title = "자동 소유권 (Auto-Owner)",
                Content = "물건이나 사람을 잡고 V키를 누르면 자동 소유권이 켜집니다. 다시 V키를 누르면 꺼집니다."
            })
        elseif GrabTab.CreateLabel then
            GrabTab:CreateLabel("자동 소유권 (Auto-Owner) - 물건이나 사람을 잡고 V키")
        else
            GrabTab:CreateSection("V키로 자동 소유권 실행")
        end
    end)

    UserInputService.InputBegan:Connect(function(input, gameProcessed)
        if gameProcessed then return end

        if input.KeyCode == Enum.KeyCode.V then
            toggleAutoOwner()
        end
    end)

    task.spawn(function()
        while task.wait(0.03) do
            if AutoOwnerT then
                local setNetworkOwner = getAutoOwnerRemote()
                local char = LocalPlayer.Character
                local hrp = char and char:FindFirstChild("HumanoidRootPart")

                if not setNetworkOwner or not hrp then
                    continue
                end

                local currentTarget, currentGrabPart = findAutoOwnerTargetPart()
                if currentTarget then
                    if currentTarget ~= AutoOwnerTargetPart then
                        if AutoOwnerHighlight and AutoOwnerHighlight.Parent then
                            AutoOwnerHighlight:Destroy()
                        end

                        AutoOwnerTargetPart = currentTarget
                        AutoOwnerGrabPart = currentGrabPart
                        AutoOwnerHighlight = createAutoOwnerHighlight(AutoOwnerTargetPart)

                        if AutoOwnerDestroyConn then
                            AutoOwnerDestroyConn:Disconnect()
                            AutoOwnerDestroyConn = nil
                        end

                        AutoOwnerDestroyConn = AutoOwnerTargetPart.Destroying:Connect(function()
                            cleanupAutoOwner()
                        end)
                    end
                end

                if not AutoOwnerTargetPart or not AutoOwnerTargetPart.Parent then
                    cleanupAutoOwner()
                    continue
                end
                pcall(function()
                    setNetworkOwner:FireServer(AutoOwnerTargetPart, AutoOwnerTargetPart.CFrame)
                end)
            end
        end
    end)

local function sitAndJumpWarp(seat)
    if not seat or not getRoot() or not getHum() then return end

    saveOriginalPosition()

    getRoot().CFrame = seat.CFrame + Vector3.new(0, 0.5, 0)

    local success = pcall(function()
        seat:Sit(getHum())
    end)

    if success then
        task.wait(0.1)

        if getHum() then
            getHum():ChangeState(Enum.HumanoidStateType.Jumping)
            returnToOriginalPosition()
        end
    else
        returnToOriginalPosition()
    end
end

    end

    local hrp = char:WaitForChild("HumanoidRootPart", 5)
    if not hum or not hrp then return end

    local firePart = hrp:FindFirstChild("FirePlayerPart")
    if firePart then firePart:Destroy() end

    hum.Changed:Connect(function(prop)
        if AntiActionsEnabled and prop == "Sit" and hum.Sit == true then
            if hum.SeatPart then
                local seatParent = hum.SeatPart.Parent
                if seatParent and tostring(seatParent) == "CreatureBlobman" then
                    return
                end
            end
            hum:SetStateEnabled(Enum.HumanoidStateType.Jumping, true)
            hum.Sit = false
        end
    end)

AntiTab:CreateSection("네트워크 파괴 방지")

do
    local Players = game:GetService("Players")
    local Workspace = game:GetService("Workspace")
    local RunService = game:GetService("RunService")
    local ReplicatedStorage = game:GetService("ReplicatedStorage")
    local LocalPlayer = Players.LocalPlayer

    local GrabEventsObj = ReplicatedStorage:FindFirstChild("GrabEvents")
    local EndGrabRemote = GrabEventsObj and GrabEventsObj:FindFirstChild("EndGrabEarly")

    local targetPos = CFrame.new(0, -9.9e9, 0)
    local monitoring = false
    local descendantConnection = nil

    local function UltraLoopProcess(item)
        if not item or not item:IsA("Instance") then return end

        local myToyFolder = Workspace:FindFirstChild(LocalPlayer.Name .. "SpawnedInToys")
        if myToyFolder and item:IsDescendantOf(myToyFolder) then
            return
        end

        task.spawn(function()
            local hp = item:FindFirstChild("HoldPart") or item:WaitForChild("HoldPart", 2)
            if not hp then return end

            local g = hp:WaitForChild("HoldItemRemoteFunction", 3)
            local d = hp:WaitForChild("DropItemRemoteFunction", 3)

            if g and d then

                while monitoring and item.Parent and hp.Parent == item do
                    local char = LocalPlayer.Character
                    if char then

                        task.spawn(function()
                            pcall(function()
                                g:InvokeServer(item, char)
                            end)
                        end)

                        task.wait(0.05)

                        task.spawn(function()
                            for i = 1, 2 do
                                pcall(function()
                                    d:InvokeServer(item, targetPos, targetPos)
                                end)
                            end
                        end)

                        if EndGrabRemote then if EndGrabRemote then EndGrabRemote:FireServer() end end
                    end

                    task.wait(0.1)
                end
            end
        end)
    end

    local function ScanAndProcess()
        for _, obj in ipairs(Workspace:GetDescendants()) do
            if obj.Name == "HoldPart" and obj.Parent then
                UltraLoopProcess(obj.Parent)
            end
        end
    end

    AntiTab:CreateToggle({
        Name = "네트워크 파괴 방지",
        CurrentValue = false,
        Flag = "AntiNetworkBreak",
        Callback = function(Value)
            monitoring = Value

            if descendantConnection then
                descendantConnection:Disconnect()
                descendantConnection = nil
            end

            if monitoring then
                task.spawn(ScanAndProcess)

                descendantConnection = Workspace.DescendantAdded:Connect(function(obj)
                    if obj.Name == "HoldPart" then
                        task.defer(function()
                            if obj.Parent and monitoring then
                                UltraLoopProcess(obj.Parent)
                            end
                        end)
                    end
                end)
            end
        end
    })
end

local function setRagdollF(state)
    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if hrp then
        local remote = ReplicatedStorage:FindFirstChild("RagdollRemote", true) or
                       (ReplicatedStorage:FindFirstChild("CharacterEvents") and ReplicatedStorage.CharacterEvents:FindFirstChild("RagdollRemote"))

        if remote then

            remote:FireServer(hrp, state and 1 or 0)
        end
    end
end

local function permRagdollLoopF()
    if permRagdollRunningS then return end
    permRagdollRunningS = true

    while permRagdollT do
        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local hum = char and char:FindFirstChild("Humanoid")

        if hrp and hum then

            setRagdollF(true)

            local moveDirection = hum.MoveDirection
            if moveDirection.Magnitude > 0 then

                hrp.AssemblyLinearVelocity = Vector3.new(
                    moveDirection.X * RAGDOLL_SPEED,
                    hrp.AssemblyLinearVelocity.Y,
                    moveDirection.Z * RAGDOLL_SPEED
                )
            end
        end

        task.wait()
    end

    permRagdollRunningS = false
    setRagdollF(false)
end

local AntiExplosionEnabled = false
local antiExplosionConn = nil

local function setupAntiExplosion(character)
    if not AntiExplosionEnabled then return end
    local hum = character:WaitForChild("Humanoid", 5)
    local partOwner = hum and hum:FindFirstChild("Ragdolled")

    if partOwner then
        if antiExplosionConn then antiExplosionConn:Disconnect() end
        antiExplosionConn = partOwner:GetPropertyChangedSignal("Value"):Connect(function()
            if not AntiExplosionEnabled then return end
            if partOwner.Value then
                for _, part in ipairs(character:GetChildren()) do
                    if part:IsA("BasePart") then
                        part.Anchored = true
                    end
                end
            else
                for _, part in ipairs(character:GetChildren()) do
                    if part:IsA("BasePart") then
                        part.Anchored = false
                    end
                end
            end
        end)
    end
end

AntiTab:CreateToggle({
    Name = "폭발 방지",
    CurrentValue = false,
    Flag = "AntiExplosion",
    Callback = function(Value)
        AntiExplosionEnabled = Value
        if Value then
            if LocalPlayer.Character then
                setupAntiExplosion(LocalPlayer.Character)
            end

            local respawnConn = LocalPlayer.CharacterAdded:Connect(setupAntiExplosion)

            task.spawn(function()
                repeat task.wait() until not AntiExplosionEnabled
                respawnConn:Disconnect()
                if antiExplosionConn then antiExplosionConn:Disconnect() end

                if LocalPlayer.Character then
                    for _, part in ipairs(LocalPlayer.Character:GetChildren()) do
                        if part:IsA("BasePart") then part.Anchored = false end
                    end
                end
            end)
        else
            if antiExplosionConn then antiExplosionConn:Disconnect() end
        end
    end
})

local antiRagdollEnabled = false
local protectionConnections = {}

local function clearProtectionConnections()
    for _, conn in ipairs(protectionConnections) do
        conn:Disconnect()
    end
    protectionConnections = {}
end

local function applyHumanoidProtection(humanoid)
    if not humanoid then return end

    humanoid.BreakJointsOnDeath = false
    humanoid.AutoRotate = true
    humanoid.PlatformStand = false

    table.insert(protectionConnections, humanoid.HealthChanged:Connect(function(health)
        if antiRagdollEnabled and health <= 0 then
            humanoid.Health = 1
        end
    end))

    table.insert(protectionConnections, humanoid:GetPropertyChangedSignal("AutoRotate"):Connect(function()
        if antiRagdollEnabled and humanoid.AutoRotate == false then
            humanoid.AutoRotate = true
        end
    end))

    table.insert(protectionConnections, humanoid:GetPropertyChangedSignal("PlatformStand"):Connect(function()
        if antiRagdollEnabled and humanoid.PlatformStand == true then
            humanoid.PlatformStand = false
        end
    end))

    table.insert(protectionConnections, RunService.RenderStepped:Connect(function()
        if antiRagdollEnabled then
            if humanoid.Sit and humanoid.SeatPart == nil then
                humanoid.Sit = false
            end
        end
    end))
end

local function onCharacterAddedProtection(char)
    local humanoid = char:WaitForChild("Humanoid", 5)
    if humanoid and antiRagdollEnabled then
        applyHumanoidProtection(humanoid)
    end
end

local function ExecuteFastGrabLoop()
    if isGrabbingLoop and _G.GrabLoopRunning then return end

    local creature = getBlobman()
    if not creature then return end

    local scripts = creature:FindFirstChild("BlobmanSeatAndOwnerScript")
    local animations = creature:FindFirstChild("BlobmanAnimations")
    local relayAnim = animations and animations:FindFirstChild("RelayClientAnimation")

    local grabEvent = scripts and scripts:FindFirstChild("CreatureGrab")
    local releaseEvent = scripts and scripts:FindFirstChild("CreatureRelease")
    local dropEvent = scripts and scripts:FindFirstChild("CreatureDrop")

    local grabEventsFolder = game:GetService("ReplicatedStorage"):FindFirstChild("GrabEvents")
    local setNetworkOwner = grabEventsFolder and grabEventsFolder:FindFirstChild("SetNetworkOwner")
    local destroyGrabLine = grabEventsFolder and grabEventsFolder:FindFirstChild("DestroyGrabLine")

    local detector = creature:FindFirstChild("RightDetector")
    local weld = detector and detector:FindFirstChild("RightWeld")

    if not (grabEvent and releaseEvent and detector and weld) then return end

    isGrabbingLoop = true
    _G.GrabLoopRunning = true

    task.spawn(function()
        local lastChars = {}

        if relayAnim then
            relayAnim:FireServer("RightGrabAnimation", true)
        end

        while isGrabbingLoop do
            if #selectedPlayers == 0 then
                RunService.Heartbeat:Wait()
                continue
            end

            for _, name in pairs(selectedPlayers) do
                if not isGrabbingLoop then break end

                local target = Players:FindFirstChild(name)
                local myRoot = getRoot()

                if target and target.Character and myRoot then
                    local tRoot = target.Character:FindFirstChild("HumanoidRootPart")
                    local tHum = target.Character:FindFirstChildOfClass("Humanoid")

                    if tRoot and tHum then

                        if target.Character ~= lastChars[name] then
                            lastChars[name] = target.Character
                            myRoot.CFrame = tRoot.CFrame * CFrame.new(0, 2, 2)

                            if setNetworkOwner then
                                setNetworkOwner:FireServer(tRoot, tRoot.CFrame)
                            end
                            task.wait(0.1)
                        end

                        pcall(function()
                            tHum.WalkSpeed = targetWalkSpeed
                            tHum.JumpPower = targetJumpPower
                            tRoot.Anchored = true

                            grabEvent:FireServer(detector, tRoot, weld)

                            releaseEvent:FireServer(weld, tRoot)

                            if dropEvent and math.random(1, 5) == 1 then
                                relayAnim:FireServer("RightTossAnimation", true)
                                dropEvent:FireServer(weld)
                                if destroyGrabLine then
                                    destroyGrabLine:FireServer(tRoot)
                                end
                            end
                        end)
                    end
                end
            end
            RunService.Heartbeat:Wait()
        end

        if relayAnim then
            relayAnim:FireServer("RightGrabAnimation", false)
        end
        UnanchorAllTargets()
        _G.GrabLoopRunning = false
    end)
end

local RespawnWatchers = {}

local function GetValidatedTarget(targetPlayer)
    if not targetPlayer or not targetPlayer.Character then return nil end

    local char = targetPlayer.Character

    local hrp = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    local head = char:FindFirstChild("Head")

    local myChar = LocalPlayer.Character
    local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")

    if not (hrp and hum and myHrp) then return nil end

    if hum:GetState() == Enum.HumanoidStateType.Dead then
        return nil
    end

    local inPlotValue = targetPlayer:FindFirstChild("InPlot")
    if inPlotValue and inPlotValue.Value == true then
        return nil
    end

    return {
        Character = char,
        Root = hrp,
        Humanoid = hum,
        Head = head or hrp,
        MyRoot = myHrp
    }
end

local function ExecutePerfectKillAttack(targetPlayer)
    local data = GetValidatedTarget(targetPlayer)
    if not data then return end

    local myChar = LocalPlayer.Character
    local myHand = myChar and (myChar:FindFirstChild("Right Arm") or myChar:FindFirstChild("RightHand"))
    if not myHand then return end

    local originalPos = myChar:GetPivot()

    task.spawn(function()
        ReplicatedStorage.GrabEvents.SetNetworkOwner:FireServer(data.Root, data.Root.CFrame)

        local startTime = tick()
        local grabSuccess = false

        myChar:PivotTo(data.Root.CFrame * CFrame.new(0, 0, 2))

        while tick() - startTime < 0.5 do
            if not data.Root or data.Humanoid:GetState() == Enum.HumanoidStateType.Dead then break end

            data.Root.Anchored = true
            data.Root.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            data.Root.AssemblyAngularVelocity = Vector3.new(0, 0, 0)

            ReplicatedStorage.GrabEvents.SetNetworkOwner:FireServer(data.Root, data.Root.CFrame)
            ReplicatedStorage.GrabEvents.CreateGrabLine:FireServer(data.Root, myHand)

            if data.Character:FindFirstChild("Head") and data.Character.Head:FindFirstChild("PartOwner") then
                grabSuccess = true

            end

            RunService.Heartbeat:Wait()
        end

        if data.Root and data.Humanoid:GetState() ~= Enum.HumanoidStateType.Dead then
            KillPlayer(targetPlayer.Name)

            for i = 1, 5 do
                ReplicatedStorage.GrabEvents.SetNetworkOwner:FireServer(data.Root, data.Root.CFrame)
                ReplicatedStorage.GrabEvents.CreateGrabLine:FireServer(data.Root, myHand)
                RunService.Heartbeat:Wait()
            end
        end

        ReplicatedStorage.GrabEvents.DestroyGrabLine:FireServer(data.Root)
        if data.Root then data.Root.Anchored = false end

        task.wait(0.05)
        if myChar:FindFirstChildOfClass("Humanoid") and myChar.Humanoid:GetState() ~= Enum.HumanoidStateType.Dead then
            myChar:PivotTo(originalPos)
        end
    end)
end

    if RespawnWatchers[targetPlayer] then RespawnWatchers[targetPlayer]:Disconnect() end

    RespawnWatchers[targetPlayer] = targetPlayer.CharacterAdded:Connect(function(char)
        if not UltimateKillEnabled then return end

        local isSelected = false
        for _, name in pairs(selectedPlayers) do
            if name == targetPlayer.Name then isSelected = true break end
        end

        if isSelected then

            local hrp = char:WaitForChild("HumanoidRootPart", 5)
            if hrp then
                task.wait()
                ExecutePerfectKillAttack(targetPlayer)
            end
        end
    end)
end

do
    local Players = game:GetService("Players")
    local ReplicatedStorage = game:GetService("ReplicatedStorage")
    local LocalPlayer = Players.LocalPlayer

    local function KillPlayer(targetPlayer)
        if not targetPlayer or not targetPlayer.Character then return end
        local humanoid = targetPlayer.Character:FindFirstChildOfClass("Humanoid")
        if humanoid then
            pcall(function()
                humanoid.BreakJointsOnDeath = false
                humanoid:ChangeState(Enum.HumanoidStateType.Dead)
            end)
        end
    end

    local function GrabRelease(blobman, targetRoot)
        pcall(function()
            local detector = blobman:FindFirstChild("LeftDetector")
            local weld = detector and detector:FindFirstChild("LeftWeld")
            local remoteScript = blobman:FindFirstChild("BlobmanSeatAndOwnerScript")

            if remoteScript and detector and weld then

                remoteScript.CreatureGrab:FireServer(detector, targetRoot, weld)

                task.wait(0.01)

                remoteScript.CreatureRelease:FireServer(weld)
            end
        end)
    end

    local function ProcessPlayer(targetPlayer)
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp or not hum then return end

        local targetChar = targetPlayer.Character
        local targetRoot = targetChar and targetChar:FindFirstChild("HumanoidRootPart")
        if not targetRoot then return end

        local toyFolder = workspace:FindFirstChild(LocalPlayer.Name .. "SpawnedInToys")
        local creature = toyFolder and toyFolder:FindFirstChild("CreatureBlobman")

        local function ExecuteGrab(blobmanInstance)
            local seat = blobmanInstance:WaitForChild("VehicleSeat", 5)
            if not seat then return end

            if hum.SeatPart ~= seat then
                hrp.CFrame = seat.CFrame
                task.wait(0.1)
                seat:Sit(hum)

                local t = tick()
                while hum.SeatPart ~= seat and tick() - t < 1.5 do
                    task.wait()
                end
            end

            if hum.SeatPart == seat then

                hrp.CFrame = targetRoot.CFrame
                task.wait(0.05)

                KillPlayer(targetPlayer)

                for i = 1, 10 do
                    GrabRelease(blobmanInstance, targetRoot)
                    task.wait(0.02)
                end
            end
        end

        if not creature then

            local connection
            connection = toyFolder.ChildAdded:Connect(function(child)
                if child.Name == "CreatureBlobman" then
                    connection:Disconnect()
                    ExecuteGrab(child)
                end
            end)

            pcall(function()
                ReplicatedStorage.MenuToys.SpawnToyRemoteFunction:InvokeServer("CreatureBlobman", hrp.CFrame, Vector3.new(0, 0, 0))
            end)
        else

            ExecuteGrab(creature)
        end
    end

    LoopTab:CreateButton({
        Name = "블롭맨 킬",
        Callback = function()
            if not selectedPlayers or #selectedPlayers == 0 then return end
            for _, playerName in pairs(selectedPlayers) do
                local target = Players:FindFirstChild(playerName)
                if target then
                    ProcessPlayer(target)
                end
            end
        end
    })
end

local function ExecuteLoopKillAttack(TargetPlayer)
    local TargetChar = TargetPlayer.Character
    local TargetRoot = TargetChar and TargetChar:FindFirstChild("HumanoidRootPart")
    local TargetHum = TargetChar and TargetChar:FindFirstChildOfClass("Humanoid")
    local TargetHead = TargetChar and TargetChar:FindFirstChild("Head")

    if TargetRoot and TargetHum and TargetHum.Health > 0 and TargetHead then
        if TargetHead:FindFirstChild("PartOwner") then
            local MyChar = LocalPlayer.Character
            local MyRoot = MyChar and MyChar:FindFirstChild("HumanoidRootPart")
            if not MyRoot then return end

            pcall(function()
                local rayOrigin = TargetRoot.Position + Vector3.new(0, 5, 0)
                local rayDirection = Vector3.new(0, -25, 0)
                local raycastParams = RaycastParams.new()
                raycastParams.FilterDescendantsInstances = {TargetChar, MyChar, Workspace.CurrentCamera}
                raycastParams.FilterType = Enum.RaycastFilterType.Exclude

                local raycastResult = Workspace:Raycast(rayOrigin, rayDirection, raycastParams)
                local teleportPos
                if raycastResult then
                    teleportPos = raycastResult.Position + Vector3.new(0, 3.5, 0)
                else
                    teleportPos = TargetRoot.Position + Vector3.new(3, 0, 0)
                end

                MyRoot.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                TargetRoot.AssemblyLinearVelocity = Vector3.new(0, 0, 0)

                MyChar:PivotTo(CFrame.new(teleportPos))

                ReplicatedStorage.GrabEvents.SetNetworkOwner:FireServer(TargetRoot, TargetRoot.CFrame)
                task.wait(0.05)
                ReplicatedStorage.GrabEvents.DestroyGrabLine:FireServer(TargetRoot)

                MyRoot.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                TargetRoot.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            end)
        end
    end
end

local function CreateSnapline(player)
    if not DrawingAvailable then
        return {Visible = false, Remove = function() end}
    end
    local line = SafeDrawingNew("Line")
    line.Visible = false
    line.Color = Box_Color
    line.Thickness = 1
    line.Transparency = 1

    local connection
    connection = RunService.RenderStepped:Connect(function()
        if Snaplines_Enabled and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            local rootPos = player.Character.HumanoidRootPart.Position
            local vector, onScreen = Camera:WorldToViewportPoint(rootPos)
            local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
            line.From = screenCenter
            if onScreen then
                line.To = Vector2.new(vector.X, vector.Y)
                line.Visible = true
            else
                local targetPos2D = Vector2.new(vector.X, vector.Y)
                local delta = (targetPos2D - screenCenter)
                local direction = (delta.Magnitude > 0 and delta.Unit) or Vector2.new(0, -1)
                if vector.Z < 0 then direction = -direction end
                local edgeX = Camera.ViewportSize.X / 2 - 15
                local edgeY = Camera.ViewportSize.Y / 2 - 15
                local m = math.min(math.abs(edgeX / direction.X), math.abs(edgeY / direction.Y))
                line.To = screenCenter + (direction * m)
                line.Visible = true
            end
        else
            line.Visible = false
            if not player or not player.Parent then pcall(function() line:Remove() end) connection:Disconnect() end
        end
    end)
    return line
end

local function ApplyESP(player)
    if player == LocalPlayer then return end
    local function setup(char)
        if not char then return end

        local highlight = Instance.new("Highlight")
        highlight.Name = "ArkadiaESP"
        highlight.FillTransparency = 0.5
        highlight.OutlineTransparency = 0
        highlight.FillColor = Box_Color
        highlight.Enabled = ESP_Enabled
        highlight.Parent = char

        local bill = CreateBox(char)
        local stroke = bill:FindFirstChild("BoxStroke", true)
        if stroke then stroke.Transparency = Box_Enabled and 0 or 1 end
        local lbl = bill:FindFirstChild("DistanceLabel", true)
        if lbl then lbl.Visible = Distance_Enabled end

        local line = CreateSnapline(player)
        ESP_Objects[player] = {Highlight = highlight, Billboard = bill, Character = char, Tracer = line}
    end
    player.CharacterAdded:Connect(setup)
    if player.Character then setup(player.Character) end
end

GrabConnections = {}
GrabToggles = {
    KillGrab = false,
    UltraGrab = false
}

function handleGrabPart(model, mode)
    if model.Name == "GrabParts" then
        local grabPart = model:FindFirstChild("GrabPart")
        local dragPart = model:FindFirstChild("DragPart")

        if mode == "울트라" and dragPart then
            pcall(function()
                dragPart.Color = Color3.fromRGB(255, 0, 0)
                dragPart.Transparency = 0
                dragPart.Material = Enum.Material.Neon
                local orientation = dragPart:FindFirstChildOfClass("AlignOrientation")
                if orientation then
                    orientation.Responsiveness = 200
                    orientation.MaxTorque = math.huge
                end
                local position = dragPart:FindFirstChildOfClass("AlignPosition")
                if position then
                    position.MaxAxesForce = Vector3.new(math.huge, math.huge, math.huge)
                    position.MaxForce = math.huge
                    position.Responsiveness = 200
                end
            end)
        end

        if mode == "Kill" and grabPart then
            local weld = grabPart:FindFirstChild("WeldConstraint")
            if weld and weld.Part1 and weld.Part1.Parent then
                local char = weld.Part1.Parent
                local hum = char:FindFirstChildOfClass("Humanoid")
                if hum then
                    hum.BreakJointsOnDeath = false
                    hum:ChangeState(Enum.HumanoidStateType.Dead)
                end
            end
        end
    end
end

function toggleGrab(flag, mode)
    if GrabConnections[flag] then
        GrabConnections[flag]:Disconnect()
        GrabConnections[flag] = nil
    end
    if GrabToggles[flag] then
        GrabConnections[flag] = Workspace.ChildAdded:Connect(function(model)
            handleGrabPart(model, mode)
        end)
    end
end

ZKeyTeleportEnabled = true
ZKeyTeleportMouse = LocalPlayer:GetMouse()
ZKeyTeleportMaxDistance = 3000

function getZTeleportRayResult()
    local char = LocalPlayer.Character
    local camera = workspace.CurrentCamera
    if not char or not camera or not ZKeyTeleportMouse then return nil end

    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    rayParams.FilterDescendantsInstances = {char, camera}
    rayParams.IgnoreWater = false

    local unitRay = camera:ScreenPointToRay(ZKeyTeleportMouse.X, ZKeyTeleportMouse.Y)
    local result = workspace:Raycast(unitRay.Origin, unitRay.Direction * ZKeyTeleportMaxDistance, rayParams)
    if result and result.Instance then
        return result
    end

    if ZKeyTeleportMouse.Target and ZKeyTeleportMouse.Hit then
        return {
            Instance = ZKeyTeleportMouse.Target,
            Position = ZKeyTeleportMouse.Hit.Position,
            Normal = Vector3.new(0, 1, 0)
        }
    end

    return nil
end

function getSafeZTeleportCFrame()
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not char or not root then return nil end

    local result = getZTeleportRayResult()
    if not result or not result.Instance then
        return nil
    end

    local standOffset = 3
    if hum then
        standOffset = math.max(3, hum.HipHeight + 2)
    end

    local normal = result.Normal or Vector3.new(0, 1, 0)
    local targetPosition

    if normal.Y > 0.25 then
        targetPosition = result.Position + Vector3.new(0, standOffset, 0)
    else
        targetPosition = result.Position + (normal * 4) + Vector3.new(0, 2, 0)
    end

    local lookVector = root.CFrame.LookVector
    local flatLook = Vector3.new(lookVector.X, 0, lookVector.Z)
    if flatLook.Magnitude < 0.01 then
        flatLook = Vector3.new(0, 0, -1)
    end

    return CFrame.lookAt(targetPosition, targetPosition + flatLook)
end

game:GetService("UserInputService").InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end
    if not ZKeyTeleportEnabled then return end
    if input.KeyCode ~= Enum.KeyCode.Z then return end

    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local targetCFrame = getSafeZTeleportCFrame()

    if char and root and targetCFrame then
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
        char:PivotTo(targetCFrame)
    end
end)

DETECT_RANGE = 25
isRunning = false
whiteFriend = false
currentMode = ""
loopConnection = nil
processingPlayers = {}

NET_INTERVAL = 0.005
NET_COUNT = 10
orbitRadius = 10
orbitSpeed = 5

function FireNetwork(part)
    if not part then return end
    pcall(function()
        local grabEvents = game:GetService("ReplicatedStorage"):FindFirstChild("GrabEvents")
        if grabEvents and grabEvents:FindFirstChild("SetNetworkOwner") then
            grabEvents.SetNetworkOwner:FireServer(part, CFrame.new(part.Position))
        end
    end)
end

function GetNearbyPlayer()
    local localChar = game.Players.LocalPlayer.Character
    if not localChar or not localChar:FindFirstChild("HumanoidRootPart") then return nil end
    local localRoot = localChar.HumanoidRootPart

    for _, player in ipairs(game.Players:GetPlayers()) do
        if player == game.Players.LocalPlayer then continue end
        if not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then continue end

        if whiteFriend and game.Players.LocalPlayer:IsFriendsWith(player.UserId) then continue end

        local targetRoot = player.Character.HumanoidRootPart
        local distance = (localRoot.Position - targetRoot.Position).Magnitude

        if distance <= DETECT_RANGE then
            return player, targetRoot
        end
    end
    return nil
end

function StartDetection(mode)
    currentMode = mode
    if loopConnection then loopConnection:Disconnect() end

    loopConnection = game:GetService("RunService").Heartbeat:Connect(function()
        if not isRunning then return end

        local player, targetRoot = GetNearbyPlayer()

        if player and not processingPlayers[player.Name] then
            processingPlayers[player.Name] = true

            task.spawn(function()
                local currentAngle = 0

                local isInfinite = (mode == "위로 잡기")
                local iterations = isInfinite and 999999 or 50

                for i = 1, iterations do

                    if not isRunning or currentMode ~= mode then break end
                    if not targetRoot or not targetRoot.Parent then break end

                    FireNetwork(targetRoot)

                    if mode == "위로 잡기" then

                        targetRoot.CFrame = targetRoot.CFrame * CFrame.new(0, 1.2, 0)
                    elseif mode == "튕김" then

                        if i == 10 then
                            targetRoot.CFrame = CFrame.new(0, 1000000000000, 0)
                            break
                        end
                    elseif mode == "궤도" then

                        currentAngle = currentAngle + math.rad(orbitSpeed)
                        local selfRoot = game.Players.LocalPlayer.Character.HumanoidRootPart
                        local offset = Vector3.new(
                            math.cos(currentAngle) * orbitRadius,
                            2,
                            math.sin(currentAngle) * orbitRadius
                        )
                        targetRoot.CFrame = CFrame.new(selfRoot.Position + offset)
                    elseif mode == "일반" then

                        targetRoot.CFrame = CFrame.new(targetRoot.Position)
                    end

                    task.wait(NET_INTERVAL)
                end

                processingPlayers[player.Name] = nil
            end)
        end
    end)
end

function SetupPhysics(obj, list)
    for _, v in pairs(list) do if v == obj then return end end
    local mainPart = obj:IsA("BasePart") and obj or obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart", true)

    if mainPart then
        pcall(function()
            if mainPart:CanSetNetworkOwnership() then
                mainPart:SetNetworkOwner(LocalPlayer)
            end
        end)

        mainPart.Anchored = false

        local bp = mainPart:FindFirstChild("ToyBodyPos") or Instance.new("BodyPosition", mainPart)
        bp.Name = "ToyBodyPos"
        bp.MaxForce = Vector3.new(1, 1, 1) * 1e8
        bp.P, bp.D = 100000, 800

        local bg = mainPart:FindFirstChild("ToyBodyGyro") or Instance.new("BodyGyro", mainPart)
        bg.Name = "ToyBodyGyro"
        bg.MaxTorque = Vector3.new(1, 1, 1) * 1e8
        bg.P = 50000

        for _, p in pairs(obj:GetDescendants()) do
            if p:IsA("BasePart") then p.CanCollide = false end
        end
        table.insert(list, obj)
    end
end

function GetShapeOffset(index, total, t, cfg)
    local h, r, speed = cfg.Height, cfg.Radius, cfg.Speed
    local angle = (index * (math.pi * 2) / total) + (t * speed)

    if cfg.CurrentShape == "행성" then
        local coreCount = math.clamp(math.floor(total * 0.35), 8, 15)
        if total <= coreCount then coreCount = total end

        local remaining = total - coreCount
        local ring1Count = math.floor(remaining / 2)

        if index <= coreCount then
            local phi = math.acos(1 - 2 * (index / coreCount))
            local theta = index * math.pi * (3 - math.sqrt(5)) + (t * speed)
            local coreR = r * 0.35
            return Vector3.new(
                math.cos(theta) * math.sin(phi) * coreR,
                math.cos(phi) * coreR + h,
                math.sin(theta) * math.sin(phi) * coreR
            )
        elseif index <= (coreCount + ring1Count) then
            local i2 = index - coreCount
            local angle2 = (i2 * (math.pi * 2) / ring1Count) + (t * speed)
            local ringR, tilt = r * 0.9, math.rad(30)
            return Vector3.new(math.cos(angle2) * ringR, math.sin(angle2) * ringR * math.sin(tilt) + h, math.sin(angle2) * ringR * math.cos(tilt))
        else
            local i3 = index - (coreCount + ring1Count)
            local ring2Count = total - (coreCount + ring1Count)
            local angle3 = (i3 * (math.pi * 2) / ring2Count) + (t * (speed * 0.8))
            local ringR, tilt = r * 1.3, math.rad(-40)
            return Vector3.new(math.cos(angle3) * ringR, math.sin(angle3) * ringR * math.sin(tilt) + h, math.sin(angle3) * ringR * math.cos(tilt))
        end

    elseif cfg.CurrentShape == "토네이도" then
        local y_pos = (index / total) * (r * 2) - r
        local current_r = ((y_pos + r) / (r * 2)) * r + 2
        return Vector3.new(math.cos(angle + (y_pos * 0.5)) * current_r, h + y_pos, math.sin(angle + (y_pos * 0.5)) * current_r)

    elseif cfg.CurrentShape == "원기둥" then
        local y_offset = (index / total) * (r * 1.5) - (r * 0.75)
        return Vector3.new(math.cos(angle) * r, h + y_offset, math.sin(angle) * r)

    elseif cfg.CurrentShape == "하트" then
        local x = 16 * math.sin(angle)^3
        local z = -(13 * math.cos(angle) - 5 * math.cos(2*angle) - 2 * math.cos(3*angle) - math.cos(4*angle))
        local pulse = 1 + 0.1 * math.sin(t * 4)
        return Vector3.new(x * (r/15) * pulse, h, z * (r/15) * pulse)
    end

    return Vector3.new(math.cos(angle) * r, h, math.sin(angle) * r)
end

function SNOWship(t)
    local hrp = getRoot()
    if hrp and t then
        do local ge = ReplicatedStorage:FindFirstChild("GrabEvents"); local r = ge and ge:FindFirstChild("SetNetworkOwner"); if r then r:FireServer(t, CFrame.lookAt(hrp.Position, t.Position)) end end
    end
end

function CheckNetworkOwnerShipOnPlayer(p)
    if p.Character and p.Character:FindFirstChild("Head") and p.Character.Head:FindFirstChild("PartOwner") then
        return p.Character.Head.PartOwner.Value == LocalPlayer.Name
    end
    return false
end

function CreateBringBody(h, c)
    h.CFrame = c
end

function CreateSkyVelocity(p)
    if p then
        local v = Instance.new("BodyVelocity")
        v.Velocity = Vector3.new(0, 10000, 0)
        v.MaxForce = Vector3.new(9e9, 9e9, 9e9)
        v.Parent = p
        game:GetService("Debris"):AddItem(v, 0.5)
    end
end

function TeleportPlayer(c)
    local hrp = getRoot()
    if hrp then hrp.CFrame = c end
end
